```swift
// Sources/OpenWisprLib/TextPipeline.swift
import Foundation

/// Pure text pipeline: prompt-assembly hints → process() → applyStyle().
/// Same code the app calls at finalizeRecording AND tests call. The extraction
/// is what prevents the inline-closure drift that shipped the comma feedback loop.
public enum TextPipeline {
    public struct Input {
        public let raw: String
        public let cursorContextText: String?
        public let screenContextText: String?
        public let punctuationMode: PunctuationMode
        public let styleMode: TextPostProcessor.StyleMode
        public let glossaryWords: String?
        public init(raw: String, cursorContextText: String?, screenContextText: String?,
                    punctuationMode: PunctuationMode, styleMode: TextPostProcessor.StyleMode,
                    glossaryWords: String?) {
            self.raw = raw; self.cursorContextText = cursorContextText
            self.screenContextText = screenContextText; self.punctuationMode = punctuationMode
            self.styleMode = styleMode; self.glossaryWords = glossaryWords
        }
    }
    public struct Result {
        public let promptHints: String?    // what would be sent to Whisper's prompt
        public let processedText: String   // after TextPostProcessor.process
        public let finalText: String       // after applyStyle
    }
    public static func run(_ input: Input) -> Result {
        let hybrid = input.punctuationMode == .hybrid
        let processed = (input.punctuationMode == .off) ? input.raw
            : TextPostProcessor.process(input.raw, hybrid: hybrid)
        let final = TextPostProcessor.applyStyle(processed, mode: input.styleMode)
        return Result(promptHints: nil, processedText: processed, finalText: final)
    }
}
```

Run: `swift test --filter TextPipelineTests`
Expected: PASS.

### Task 5: Fix WhisperEngine use-after-free in streaming callback (SCOUT #2)

**File:** `Sources/OpenWisprLib/WhisperEngine.swift:209-243`

**Bug:** `callbackCtx` is `Unmanaged.passRetained` at L209, released at L243. The callback does `DispatchQueue.main.async { cbCtx.onPartialResult(...) }` at L228-230 — if main runs *after* L243's release, it derefs freed memory.

**Step 1: Add a failing test (best-effort — UAF is hard to deterministically reproduce; assert the structural fix)**

Write a unit test that creates a `WhisperEngine`, kicks off a fake streaming pass that posts a callback, releases the context immediately, and asserts no crash + the callback runs with valid data. Use stub callbacks.

**Step 2: Fix**

Before the `DispatchQueue.main.async` block, copy the closure + the partial string into local lets, capture only those, and *don't* deref `cbCtx` from inside the async closure:

```swift
let cb = cbCtx.onPartialResult
let snapshot = trimmed
DispatchQueue.main.async { cb(snapshot) }
```

### Task 10: Stale OCR generation token (SCOUT #9)

**File:** `Sources/OpenWisprLib/AppDelegate.swift:454-499, 564-567`

```swift
func test_appcastSignatureIsValidBase64() throws {
    let url = URL(fileURLWithPath: "docs/appcast.xml")
    let xml = try String(contentsOf: url)
    let match = xml.range(of: #"edSignature="([^"]+)""#, options: .regularExpression)
    XCTAssertNotNil(match, "appcast missing edSignature attribute")
    let sig = String(xml[match!]).replacingOccurrences(of: #"edSignature=""#, with: "").dropLast()
    // Real EdDSA sig: ~86-88 chars, base64. Length number alone would be all digits.
    XCTAssertGreaterThanOrEqual(sig.count, 40, "signature too short — extraction likely captured the length again")
    XCTAssertFalse(sig.allSatisfy(\.isNumber), "signature is all digits — extraction captured length=, not edSignature=")
    XCTAssertNotNil(Data(base64Encoded: String(sig)), "signature is not valid base64")
}
```
